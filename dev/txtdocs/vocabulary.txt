	if(p1!=undefined)
		slovo+=p1;

chislitlx(число, существительное) - Возвращает строку, состоящую из данного числа и подходящего падежа слова p1, при этом
полученное словосочетанию стоит в падеже p2 (если не указан - именительный).
Используем в основной массе его. 
Принцип работы:
function chislitlx(p1,p2,p3,separator){
/***/

	let options = String(p3 || '');
	let tex  = !!(1+options.search(/\$/g));
	options  =  options.replace(/\$/g, '');
	let nots = !!(1+options.search(/\./g));
	options  =  options.replace(/\./g, '');

	var rez=sklonlxkand(p2,undefined,0);
	switch(options){
		case 'r': return chislitM(p1,	rez.re,	 rez.rm,                rez.rm, tex, nots, separator);
		case 'd': return chislitM(p1,	rez.de,	 rez.dm,                rez.dm, tex, nots, separator);
		case 'v': return chislitM(p1,	rez.ve,	(rez.r2?rez.r2:rez.re),	rez.rm, tex, nots, separator);
		case 't': return chislitM(p1,	rez.te,	 rez.tm,                rez.tm, tex, nots, separator);
		case 'p': return chislitM(p1,	rez.pe,	 rez.pm,                rez.pm, tex, nots, separator);
		case 'i':
		default : return chislitM(p1,	rez.ie,	(rez.r2?rez.r2:rez.re),	rez.rm, tex, nots, separator);
	}
}

chislitM - Вспомогательная функция для согласования существительного с числительным. Используется куда реже chislitlx
function chislitM(p1,p2,p3,p4,tex,nots,separator){
	if(Object.prototype.toString.call(separator) !== '[object String]') {
		
		separator = ' ';
	}

	let number = '';
	if(tex && nots){
		number = '$' + p1 + '$';
	} else 	if(tex && !nots){
		number = '$' + p1.ts(true) + '$';
	} else 	if(!tex && nots){
		number = '' + p1;
	} else { // !tex && !nots
		number = p1.ts();
	}

	return number + separator + chislit(p1,p2,p3,p4);
}
Пример использования: 
chislitM(distanceToLeftStation*1000, 'метре', 'метрах', 'метрах')



chislit - Вспомогательная функция для согласования существительного с числительным. 
function chislit(a,s1,s2,s5){
/**Вспомогательная процедура*/
	a=a%100;

	if(!a.isAlmostInteger())
		return s2;

	if((a>=5)&&(a<=20))
		return s5;

	a=a%10;
	if(a==1)
		return s1;

	if((a<=4)&&(a>=2))
		return s2;

	return s5;
}
Пример использования:
chislit(d,'имеет','имеют','имеют')
